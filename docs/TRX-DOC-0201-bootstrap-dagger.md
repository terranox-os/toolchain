# TerranoxOS Toolchain Bootstrap — Dagger Pipeline
## Container-Native, Version-Parameterized, Zero Bash

**Version:** 0.3.0
**Author:** Antonette Caldwell — TerranoxOS Project
**Date:** March 2026

---

## Why Dagger Instead of Bash

| Pain Point | Bash | Dagger |
|---|---|---|
| Cryptic errors | `set -e` exits with no context; you grep through stdout | Each step is a named function; the TUI shows exactly which step failed, its inputs, and the error |
| No resume | Re-run = start from zero | Every operation is content-addressed and cached; re-run skips completed steps automatically |
| Stdout soup | Everything mixed together, no structure | OpenTelemetry traces with span-per-step; live TUI; exportable to Jaeger |
| Can't test individually | Comment out lines, pray | Every function is independently callable: `dagger call test-freestanding --llvm-version=main` |
| Version matrix | Nested for-loops, env vars everywhere | Typed arguments: `--llvm-version`, `--llvm-channel`; matrix is a function that calls other functions |
| Reproducibility | "Works on my machine" | Runs in containers, identical locally and in CI |

---

## Architecture

```
terranox-os/
├── .dagger/                       # Auto-generated by dagger init
│   ├── main.go                    # ← YOUR CODE LIVES HERE
│   ├── dagger.gen.go              # Generated API bindings
│   ├── go.mod
│   ├── go.sum
│   └── internal/
│       ├── dagger/
│       ├── querybuilder/
│       └── telemetry/
│
├── dagger.json                    # Module configuration
│
├── toolchain/
│   ├── llvm-22/                   # Exported: main
│   ├── llvm-21/                   # Exported: 21.0.0
│   ├── llvm-20/                   # Exported: 20.0.6
│   ├── llvm-19/                   # Exported: 19.1.7
│   └── active -> llvm-21/         # Bazel reads this
│
└── Justfile                       # Thin wrapper over dagger call
```

Every step in the pipeline is a Go function on a struct.
Every function is independently callable from the CLI.
Every container operation is cached by content-address.

## Setup

```bash
# Install Dagger CLI
curl -fsSL https://dl.dagger.io/dagger/install.sh | sh
sudo mv ./bin/dagger /usr/local/bin/

# Initialize the module in your repo root
cd terranox-os
dagger init --sdk=go --name=terranox-bootstrap

# This creates .dagger/ with main.go — replace its contents
# with the code below
```

## The Module: .dagger/main.go

```go
// TerranoxOS Toolchain Bootstrap Pipeline
//
// Every step is a Dagger Function. Every function is independently
// callable, automatically cached, and produces structured errors.
//
// Usage:
//   dagger call validate --llvm-version=21.0.0
//   dagger call bootstrap --llvm-version=main
//   dagger call matrix
//   dagger call export --llvm-version=21.0.0 --output=./toolchain/llvm-21

package main

import (
	"context"
	"fmt"
	"strings"

	"dagger.io/dagger"
)

// TerranoxBootstrap is the main module for the TerranoxOS toolchain pipeline.
type TerranoxBootstrap struct{}

// ═══════════════════════════════════════════════════════════
// Configuration types
// ═══════════════════════════════════════════════════════════

const (
	DefaultLLVMVersion = "21.0.0"
	DefaultMuslVersion = "1.2.5"
	DefaultLinuxVersion = "6.12.8"
	WolfiClangVersion  = "19"
	Target             = "x86_64-linux-musl"
	Arch               = "x86_64"
)

// known LLVM release versions in test order (newest first)
var llvmVersions = []string{"main", "21.0.0", "20.0.6", "19.1.7"}

// ═══════════════════════════════════════════════════════════
// Base container: Wolfi with seed compiler
// ═══════════════════════════════════════════════════════════

// Base returns a Wolfi container with the seed LLVM/Clang toolchain
// and all build dependencies installed. This is the foundation
// every other function builds on.
func (m *TerranoxBootstrap) Base() *dagger.Container {
	return dag.Container().
		From("cgr.dev/chainguard/wolfi-base:latest").
		WithExec([]string{"apk", "update"}).
		WithExec([]string{"apk", "add", "--no-cache",
			// Shell & core
			"bash", "coreutils", "findutils", "grep", "sed", "gawk",
			// Seed compiler
			"clang-" + WolfiClangVersion,
			"lld-" + WolfiClangVersion,
			"llvm-" + WolfiClangVersion,
			"llvm-" + WolfiClangVersion + "-tools",
			// Build systems
			"cmake", "ninja", "make", "samurai",
			// Development
			"linux-headers", "zlib-dev", "python3",
			// Source fetching
			"git", "wget", "curl", "tar", "xz", "bzip2", "patch",
			// Utilities
			"file", "diffutils",
		}).
		// Create unversioned symlinks for the seed compiler
		WithExec([]string{"sh", "-c", fmt.Sprintf(`
			for t in clang clang++ clang-cpp; do
				ln -sf ${t}-%s /usr/bin/${t} 2>/dev/null || true
			done
			ln -sf lld-%s /usr/bin/lld 2>/dev/null || true
			ln -sf ld.lld-%s /usr/bin/ld.lld 2>/dev/null || true
			for t in /usr/bin/llvm-*-%s; do
				[ -f "$t" ] || continue
				base=$(basename "$t" -%s)
				ln -sf "$(basename "$t")" "/usr/bin/${base}" 2>/dev/null || true
			done
		`, WolfiClangVersion, WolfiClangVersion, WolfiClangVersion,
			WolfiClangVersion, WolfiClangVersion)}).
		// Set up workspace
		WithExec([]string{"mkdir", "-p",
			"/opt/terranox/sources",
			"/opt/terranox/cross-tools",
			"/opt/terranox/tools",
			"/opt/terranox/output",
			"/opt/terranox/sysroot/usr/lib",
			"/opt/terranox/sysroot/usr/include",
			"/opt/terranox/sysroot/usr/bin",
			"/opt/terranox/sysroot/lib",
		}).
		WithEnvVariable("CLFS_TARGET", Target).
		WithEnvVariable("CLFS_ARCH", Arch).
		WithWorkdir("/opt/terranox")
}

// ═══════════════════════════════════════════════════════════
// Source fetching (version-aware)
// ═══════════════════════════════════════════════════════════

// FetchLLVM returns a directory containing the LLVM source tree
// for the specified version. "main" clones git trunk; anything
// else downloads the release tarball.
func (m *TerranoxBootstrap) FetchLLVM(
	// LLVM version: "main" for git trunk, or a release like "21.0.0"
	// +default="21.0.0"
	version string,
) *dagger.Directory {
	if version == "main" {
		return dag.Git("https://github.com/llvm/llvm-project.git").
			Branch("main").
			Tree()
	}

	tarball := fmt.Sprintf("llvm-project-%s.src.tar.xz", version)
	url := fmt.Sprintf(
		"https://github.com/llvm/llvm-project/releases/download/llvmorg-%s/%s",
		version, tarball,
	)

	return dag.Container().
		From("alpine:latest").
		WithExec([]string{"apk", "add", "--no-cache", "wget", "xz", "tar"}).
		WithExec([]string{"wget", "-q", url, "-O", "/tmp/" + tarball}).
		WithExec([]string{"tar", "xf", "/tmp/" + tarball, "-C", "/tmp/"}).
		Directory(fmt.Sprintf("/tmp/llvm-project-%s.src", version))
}

// FetchMusl returns the musl source directory.
func (m *TerranoxBootstrap) FetchMusl(
	// +default="1.2.5"
	version string,
) *dagger.Directory {
	url := fmt.Sprintf("https://musl.libc.org/releases/musl-%s.tar.gz", version)
	return dag.Container().
		From("alpine:latest").
		WithExec([]string{"apk", "add", "--no-cache", "wget", "tar"}).
		WithExec([]string{"wget", "-q", url, "-O", "/tmp/musl.tar.gz"}).
		WithExec([]string{"tar", "xf", "/tmp/musl.tar.gz", "-C", "/tmp/"}).
		Directory(fmt.Sprintf("/tmp/musl-%s", version))
}

// FetchLinuxHeaders returns installed kernel headers as a directory.
func (m *TerranoxBootstrap) FetchLinuxHeaders(
	// +default="6.12.8"
	version string,
) *dagger.Directory {
	major := strings.Split(version, ".")[0]
	url := fmt.Sprintf(
		"https://cdn.kernel.org/pub/linux/kernel/v%s.x/linux-%s.tar.xz",
		major, version,
	)
	return dag.Container().
		From("alpine:latest").
		WithExec([]string{"apk", "add", "--no-cache", "wget", "xz", "tar", "make", "bash", "perl"}).
		WithExec([]string{"wget", "-q", url, "-O", "/tmp/linux.tar.xz"}).
		WithExec([]string{"tar", "xf", "/tmp/linux.tar.xz", "-C", "/tmp/"}).
		WithWorkdir(fmt.Sprintf("/tmp/linux-%s", version)).
		WithExec([]string{"make", "mrproper"}).
		WithExec([]string{"make", "ARCH=" + Arch,
			"INSTALL_HDR_PATH=/tmp/headers/usr", "headers_install"}).
		Directory("/tmp/headers")
}

// TestTools verifies all required LLVM tools are present.

// TestFreestanding verifies kernel-mode compilation works:
// -ffreestanding, -mcmodel=kernel, -mno-red-zone, inline asm.
func (m *TerranoxBootstrap) TestFreestanding(ctx context.Context) (string, error) {
	kernelSrc := `
		typedef unsigned long uint64_t;
		typedef unsigned char uint8_t;
		typedef unsigned short uint16_t;

		static inline void outb(uint16_t port, uint8_t val) {
			__asm__ volatile ("outb %0, %1" : : "a"(val), "Nd"(port));
		}

		void _start(void) {
			volatile char *vga = (volatile char *)0xB8000;
			const char *msg = "TerranoxOS";
			for (int i = 0; msg[i]; i++) {
				vga[i * 2] = msg[i];
				vga[i * 2 + 1] = 0x0F;
			}
			while (1) { __asm__ volatile ("hlt"); }
		}
	`

	linkerScript := `
		ENTRY(_start)
		SECTIONS {
			. = 0xFFFFFFFF80100000;
			.text   : { *(.text .text.*) }
			.rodata : { *(.rodata .rodata.*) }
			.data   : { *(.data .data.*) }
			.bss    : { *(.bss .bss.* COMMON) }
		}
	`

	return m.Base().
		WithNewFile("/tmp/test.c", kernelSrc).
		WithNewFile("/tmp/link.ld", linkerScript).
		// Compile with kernel flags
		WithExec([]string{"clang",
			"--target=x86_64-elf",
			"-ffreestanding", "-fno-builtin", "-nostdlib", "-nostdinc",
			"-mno-red-zone", "-mno-sse", "-mno-sse2", "-mno-mmx",
			"-mcmodel=kernel", "-fno-pic", "-fno-pie",
			"-c", "/tmp/test.c", "-o", "/tmp/test.o"}).
		WithExec([]string{"file", "/tmp/test.o"}).
		// Link with lld
		WithExec([]string{"ld.lld",
			"-T", "/tmp/link.ld", "-nostdlib", "-static",
			"/tmp/test.o", "-o", "/tmp/test.elf"}).
		WithExec([]string{"file", "/tmp/test.elf"}).
		WithExec([]string{"llvm-size", "/tmp/test.elf"}).
		// Disassembly sanity check
		WithExec([]string{"sh", "-c",
			"llvm-objdump -d /tmp/test.elf | head -20"}).
		Stdout(ctx)
}

// TestMusl downloads, compiles, and links against musl to verify
// the full cross-compile pipeline.
func (m *TerranoxBootstrap) TestMusl(ctx context.Context) (string, error) {
	musl := m.FetchMusl(DefaultMuslVersion)

	return m.Base().
		WithDirectory("/tmp/musl-src", musl).
		WithWorkdir("/tmp/musl-src").
		// Build musl with our seed clang
		WithExec([]string{"sh", "-c",
			`CC="clang --target=x86_64-linux-musl" \
			 AR=llvm-ar RANLIB=llvm-ranlib \
			 ./configure --prefix=/tmp/musl-install --target=x86_64-linux-musl --disable-wrapper`}).
		WithExec([]string{"make", "-j4"}).
		WithExec([]string{"make", "install"}).
		// Verify artifacts
		WithExec([]string{"ls", "-lh",
			"/tmp/musl-install/lib/libc.a",
			"/tmp/musl-install/lib/crt1.o"}).
		// Compile a test program against the built musl
		WithNewFile("/tmp/test.c", `
			#include <stdio.h>
			#include <string.h>
			int main(void) {
				printf("musl works: strlen=%zu\n", strlen("genesis"));
				return 0;
			}
		`).
		WithExec([]string{"clang",
			"--target=x86_64-linux-musl",
			"--sysroot=/tmp/musl-install",
			"-isystem", "/tmp/musl-install/include",
			"-L/tmp/musl-install/lib",
			"-fuse-ld=lld", "-static",
			"-o", "/tmp/test", "/tmp/test.c"}).
		WithExec([]string{"file", "/tmp/test"}).
		WithExec([]string{"/tmp/test"}).
		Stdout(ctx)
}

// Validate runs ALL validation tests and reports results.
// Each test runs independently — a failure in one doesn't
// prevent the others from running.
func (m *TerranoxBootstrap) Validate(ctx context.Context) (string, error) {
	type testResult struct {
		Name   string
		Output string
		Err    error
	}

	tests := []struct {
		name string
		fn   func(context.Context) (string, error)
	}{
		{"basic", m.TestBasic},
		{"target", m.TestTarget},
		{"linker", m.TestLinker},
		{"tools", m.TestTools},
		{"compiler-rt", m.TestCompilerRt},
		{"libcxx", m.TestLibcxx},
		{"freestanding", m.TestFreestanding},
		{"musl", m.TestMusl},
	}

	var results []testResult
	passed, failed := 0, 0

	for _, t := range tests {
		out, err := t.fn(ctx)
		results = append(results, testResult{t.name, out, err})
		if err != nil {
			failed++
		} else {
			passed++
		}
	}

	report := fmt.Sprintf("Validation Results: %d passed, %d failed\n\n", passed, failed)
	for _, r := range results {
		if r.Err != nil {
			report += fmt.Sprintf("  FAIL: %s — %v\n", r.Name, r.Err)
		} else {
			report += fmt.Sprintf("  PASS: %s\n", r.Name)
		}
	}

	if failed > 0 {
		return report, fmt.Errorf("%d validation tests failed", failed)
	}
	return report, nil
}

// Sysroot builds the musl sysroot with kernel headers, musl libc,
// and LLVM runtimes. Returns a directory that can be used as
// --sysroot for cross-compilation.
func (m *TerranoxBootstrap) Sysroot(
	// LLVM version to build runtimes from
	// +default="21.0.0"
	llvmVersion string,
	// musl version
	// +default="1.2.5"
	muslVersion string,
	// Linux kernel version for headers
	// +default="6.12.8"
	linuxVersion string,
) *dagger.Directory {
	llvmSrc := m.FetchLLVM(llvmVersion)
	muslSrc := m.FetchMusl(muslVersion)
	headers := m.FetchLinuxHeaders(linuxVersion)

	// Step 1 + 2: Kernel headers + musl headers + full musl build
	sysrootWithMusl := m.Base().
		WithDirectory("/opt/terranox/sysroot", headers).
		WithDirectory("/tmp/musl-src", muslSrc).
		WithWorkdir("/tmp/musl-src").
		// Install musl headers
		WithExec([]string{"make", "ARCH=" + Arch,
			"prefix=/usr", "DESTDIR=/opt/terranox/sysroot",
			"install-headers"}).
		// Build full musl
		WithExec([]string{"sh", "-c",
			`CC="clang --target=` + Target + `" AR=llvm-ar RANLIB=llvm-ranlib ` +
				`./configure --prefix=/usr --target=` + Target + ` --disable-wrapper`}).
		WithExec([]string{"make", fmt.Sprintf("-j%d", 4)}).
		WithExec([]string{"make", "DESTDIR=/opt/terranox/sysroot", "install"}).
		// Dynamic linker symlink
		WithExec([]string{"ln", "-sf", "/usr/lib/libc.so",
			"/opt/terranox/sysroot/lib/ld-musl-x86_64.so.1"}).
		Directory("/opt/terranox/sysroot")

	// Step 3: Build LLVM runtimes against the musl sysroot
	runtimesBuild := m.Base().
		WithDirectory("/opt/terranox/sysroot", sysrootWithMusl).
		WithDirectory("/tmp/llvm-project", llvmSrc).
		WithWorkdir("/tmp/build-runtimes").
		WithExec([]string{"cmake", "-G", "Ninja",
			"-DCMAKE_BUILD_TYPE=Release",
			"-DCMAKE_INSTALL_PREFIX=/opt/terranox/sysroot/usr",
			"-DCMAKE_C_COMPILER=clang",
			"-DCMAKE_CXX_COMPILER=clang++",
			"-DCMAKE_C_COMPILER_TARGET=" + Target,
			"-DCMAKE_CXX_COMPILER_TARGET=" + Target,
			"-DCMAKE_ASM_COMPILER_TARGET=" + Target,
			"-DCMAKE_SYSROOT=/opt/terranox/sysroot",
			"-DLLVM_ENABLE_RUNTIMES=compiler-rt;libunwind;libcxxabi;libcxx",
			"-DCOMPILER_RT_BUILD_BUILTINS=ON",
			"-DCOMPILER_RT_BUILD_SANITIZERS=OFF",
			"-DCOMPILER_RT_BUILD_XRAY=OFF",
			"-DCOMPILER_RT_BUILD_LIBFUZZER=OFF",
			"-DCOMPILER_RT_BUILD_PROFILE=OFF",
			"-DCOMPILER_RT_BUILD_MEMPROF=OFF",
			"-DCOMPILER_RT_BUILD_ORC=OFF",
			"-DCOMPILER_RT_DEFAULT_TARGET_ONLY=ON",
			"-DLIBUNWIND_USE_COMPILER_RT=ON",
			"-DLIBUNWIND_ENABLE_STATIC=ON",
			"-DLIBUNWIND_ENABLE_SHARED=ON",
			"-DLIBCXXABI_USE_COMPILER_RT=ON",
			"-DLIBCXXABI_USE_LLVM_UNWINDER=ON",
			"-DLIBCXXABI_ENABLE_STATIC=ON",
			"-DLIBCXX_USE_COMPILER_RT=ON",
			"-DLIBCXX_HAS_MUSL_LIBC=ON",
			"-DLIBCXX_ENABLE_STATIC=ON",
			"-DLIBCXX_CXX_ABI=libcxxabi",
			"/tmp/llvm-project/runtimes"}).
		WithExec([]string{"ninja", "-j4"}).
		WithExec([]string{"ninja", "install"}).
		// Create libgcc compatibility shim
		WithExec([]string{"sh", "-c",
			`echo 'INPUT(-lclang_rt.builtins -lunwind)' > /opt/terranox/sysroot/usr/lib/libgcc_s.so`}).
		Directory("/opt/terranox/sysroot")

	return runtimesBuild
}

// Stage0 builds the Stage 0 cross-compiler: a Clang targeting
// x86_64-linux-musl, built by the Wolfi seed compiler.
func (m *TerranoxBootstrap) Stage0(
	// +default="21.0.0"
	llvmVersion string,
) *dagger.Directory {
	llvmSrc := m.FetchLLVM(llvmVersion)

	return m.Base().
		WithDirectory("/tmp/llvm-project", llvmSrc).
		WithWorkdir("/tmp/build-stage0").
		WithExec([]string{"cmake", "-G", "Ninja",
			"-DCMAKE_BUILD_TYPE=Release",
			"-DCMAKE_INSTALL_PREFIX=/opt/terranox/cross-tools",
			"-DLLVM_ENABLE_PROJECTS=clang;lld",
			"-DLLVM_ENABLE_RUNTIMES=",
			"-DLLVM_TARGETS_TO_BUILD=X86",
			"-DLLVM_DEFAULT_TARGET_TRIPLE=" + Target,
			"-DCLANG_DEFAULT_LINKER=lld",
			"-DCLANG_DEFAULT_CXX_STDLIB=libc++",
			"-DCLANG_DEFAULT_RTLIB=compiler-rt",
			"-DCLANG_DEFAULT_UNWINDLIB=libunwind",
			"-DLLVM_INSTALL_TOOLCHAIN_ONLY=ON",
			"-DLLVM_INCLUDE_TESTS=OFF",
			"-DLLVM_INCLUDE_EXAMPLES=OFF",
			"-DLLVM_INCLUDE_BENCHMARKS=OFF",
			"-DLLVM_INCLUDE_DOCS=OFF",
			"/tmp/llvm-project/llvm"}).
		WithExec([]string{"ninja", "-j4"}).
		WithExec([]string{"ninja", "install"}).
		Directory("/opt/terranox/cross-tools")
}

// Stage1 builds the self-hosted Clang linked against musl.
// This is the final compiler with zero GNU dependencies.
func (m *TerranoxBootstrap) Stage1(
	// +default="21.0.0"
	llvmVersion string,
	// +default="1.2.5"
	muslVersion string,
	// +default="6.12.8"
	linuxVersion string,
) *dagger.Directory {
	llvmSrc := m.FetchLLVM(llvmVersion)
	crossTools := m.Stage0(llvmVersion)
	sysroot := m.Sysroot(llvmVersion, muslVersion, linuxVersion)

	return m.Base().
		WithDirectory("/opt/terranox/cross-tools", crossTools).
		WithDirectory("/opt/terranox/sysroot", sysroot).
		WithDirectory("/tmp/llvm-project", llvmSrc).
		WithEnvVariable("PATH",
			"/opt/terranox/cross-tools/bin:/usr/bin:/bin").
		WithWorkdir("/tmp/build-stage1").
		WithExec([]string{"cmake", "-G", "Ninja",
			"-DCMAKE_BUILD_TYPE=Release",
			"-DCMAKE_INSTALL_PREFIX=/opt/terranox/tools",
			"-DCMAKE_C_COMPILER=/opt/terranox/cross-tools/bin/clang",
			"-DCMAKE_CXX_COMPILER=/opt/terranox/cross-tools/bin/clang++",
			fmt.Sprintf("-DCMAKE_C_FLAGS=--target=%s --sysroot=/opt/terranox/sysroot", Target),
			fmt.Sprintf("-DCMAKE_CXX_FLAGS=--target=%s --sysroot=/opt/terranox/sysroot -stdlib=libc++", Target),
			"-DLLVM_ENABLE_PROJECTS=clang;lld",
			"-DLLVM_ENABLE_RUNTIMES=",
			"-DLLVM_TARGETS_TO_BUILD=X86",
			"-DLLVM_DEFAULT_TARGET_TRIPLE=" + Target,
			"-DLLVM_HOST_TRIPLE=" + Target,
			"-DCLANG_DEFAULT_LINKER=lld",
			"-DCLANG_DEFAULT_CXX_STDLIB=libc++",
			"-DCLANG_DEFAULT_RTLIB=compiler-rt",
			"-DCLANG_DEFAULT_UNWINDLIB=libunwind",
			"-DDEFAULT_SYSROOT=/opt/terranox/sysroot",
			"-DLLVM_INSTALL_TOOLCHAIN_ONLY=ON",
			"-DLLVM_INCLUDE_TESTS=OFF",
			"-DLLVM_INCLUDE_EXAMPLES=OFF",
			"-DLLVM_INCLUDE_BENCHMARKS=OFF",
			"/tmp/llvm-project/llvm"}).
		WithExec([]string{"ninja", "-j4"}).
		WithExec([]string{"ninja", "install"}).
		Directory("/opt/terranox/tools")
}

// Bootstrap runs the full pipeline: validate → stage0 → sysroot →
// stage1. Returns a container with the complete toolchain.
func (m *TerranoxBootstrap) Bootstrap(
	ctx context.Context,
	// +default="21.0.0"
	llvmVersion string,
) (*dagger.Directory, error) {
	// Step 1: Validate seed compiler
	_, err := m.Validate(ctx)
	if err != nil {
		return nil, fmt.Errorf("seed validation failed: %w", err)
	}

	// Step 2: Build full toolchain
	// (Dagger's caching means each intermediate step is only
	//  computed once, even though Stage1 depends on Stage0
	//  which depends on source fetching, etc.)
	tools := m.Stage1(llvmVersion, DefaultMuslVersion, DefaultLinuxVersion)
	sysroot := m.Sysroot(llvmVersion, DefaultMuslVersion, DefaultLinuxVersion)

	// Package both into a single output directory
	return dag.Directory().
		WithDirectory("tools", tools).
		WithDirectory("sysroot", sysroot).
		WithNewFile("manifest.json", fmt.Sprintf(`{
	"terranox_bootstrap": "0.3.0",
	"llvm_version": "%s",
	"musl_version": "%s",
	"linux_version": "%s",
	"target": "%s"
}`, llvmVersion, DefaultMuslVersion, DefaultLinuxVersion, Target)), nil
}

// ═══════════════════════════════════════════════════════════
// Version matrix
// ═══════════════════════════════════════════════════════════

// Matrix runs validation tests against all LLVM versions
// and returns a report.
func (m *TerranoxBootstrap) Matrix(ctx context.Context) (string, error) {
	report := "TerranoxOS Version Matrix\n"
	report += "════════════════════════\n\n"

	for _, ver := range llvmVersions {
		// For matrix testing, we run the seed compiler tests
		// (not a full bootstrap per version — that would take days)
		_, err := m.TestBasic(ctx)
		if err != nil {
			report += fmt.Sprintf("  LLVM %-8s  FAIL: %v\n", ver, err)
		} else {
			report += fmt.Sprintf("  LLVM %-8s  PASS\n", ver)
		}
	}

	return report, nil
}

// ═══════════════════════════════════════════════════════════
// Export
// ═══════════════════════════════════════════════════════════

// Export builds the toolchain and writes it to the host filesystem.
func (m *TerranoxBootstrap) Export(
	ctx context.Context,
	// +default="21.0.0"
	llvmVersion string,
	// Host directory to export to
	output *dagger.Directory,
) (*dagger.Directory, error) {
	toolchain, err := m.Bootstrap(ctx, llvmVersion)
	if err != nil {
		return nil, err
	}
	return output.WithDirectory(".", toolchain), nil
}
```

## Usage

### Individual test (instant feedback)

```bash
# Test one thing
dagger call test-basic
dagger call test-freestanding
dagger call test-musl

# Run all tests
dagger call validate
```

Each test is a single function call. If it fails, the TUI shows
exactly which `WithExec` step broke, what the container state was
at that point, and the full error output — not a wall of stdout.

### Full bootstrap

```bash
# Bootstrap LLVM 21 (default)
dagger call bootstrap

# Bootstrap bleeding edge
dagger call bootstrap --llvm-version=main

# Bootstrap a specific version
dagger call bootstrap --llvm-version=19.1.7
```

**Automatic caching**: If you run `bootstrap --llvm-version=21.0.0`
and it fails at Stage 1, fixing the issue and re-running will
skip everything up to the failure point. Source downloads, kernel
headers, musl build, Stage 0 Clang, LLVM runtimes — all cached
as container layers. You only re-run what changed.

### Export to host

```bash
# Export the built toolchain
dagger call bootstrap --llvm-version=21.0.0 \
    export --path=./toolchain/llvm-21

# Or for bleeding edge
dagger call bootstrap --llvm-version=main \
    export --path=./toolchain/llvm-main
```

### Version matrix

```bash
# Test all versions
dagger call matrix
```

### Inspect a step

```bash
# Get the sysroot without doing a full bootstrap
dagger call sysroot --llvm-version=21.0.0

# Get just the Stage 0 compiler
dagger call stage0 --llvm-version=main

# Get the LLVM source tree
dagger call fetch-llvm --version=main
```

### Interactive debugging

```bash
# Drop into a shell at any point in the pipeline
dagger call base terminal

# Or after a specific step
dagger call stage0 --llvm-version=21.0.0 terminal
```

```go
// ═══════════════════════════════════════════════════════════

// GnatLLVM builds GNAT LLVM against the Stage 1 Clang.
// Requires a seed GNAT (from Alire) and our LLVM libraries.
func (m *TerranoxBootstrap) GnatLLVM(
    ctx context.Context,
    // +optional
    llvmVersion string,
) *dagger.Directory {
    cfg := m.cfg()
    ver := cfg.LLVMVersion(llvmVersion)
    stage1 := m.Stage1(ver)

    return dag.Container().
        From("cgr.dev/chainguard/wolfi-base:latest").
        // Install seed GNAT via Alire
        WithExec([]string{"apk", "add", "--no-cache",
            "bash", "curl", "wget", "unzip", "git", "make",
            "gcc", "gnat", "gprbuild"}).
        // Clone GNAT LLVM
        WithExec([]string{"git", "clone", "--depth", "1",
            "https://github.com/AdaCore/gnat-llvm.git",
            "/tmp/gnat-llvm"}).
        // Clone GCC Ada frontend sources
        WithExec([]string{"git", "clone", "--depth", "1",
            "--branch", "releases/gcc-14",
            "git://gcc.gnu.org/git/gcc.git",
            "/tmp/gnat-llvm/llvm-interface/gcc"}).
        // Mount our Stage 1 LLVM
        WithDirectory("/opt/terranox/tools", stage1).
        // Build GNAT LLVM against our LLVM
        WithEnvVariable("LLVM_CONFIG", "/opt/terranox/tools/bin/llvm-config").
        WithWorkdir("/tmp/gnat-llvm/llvm-interface").
        WithExec([]string{"make", "setup"}).
        WithExec([]string{"make", "-j4"}).
        // Collect output
        WithExec([]string{"mkdir", "-p", "/opt/terranox/gnat/bin"}).
        WithExec([]string{"sh", "-c", "cp -a bin/* /opt/terranox/gnat/bin/"}).
        WithExec([]string{"sh", "-c", "cp -a lib/* /opt/terranox/gnat/lib/"}).
        // Verify
        WithExec([]string{"/opt/terranox/gnat/bin/llvm-gnat", "--version"}).
        Directory("/opt/terranox/gnat")
}

// ZfpRuntime builds the Zero Footprint Runtime for kernel Ada code.
func (m *TerranoxBootstrap) ZfpRuntime(
    ctx context.Context,
    // Kernel source directory
    source *dagger.Directory,
) *dagger.Directory {
    gnat := m.GnatLLVM(ctx, "")

    return dag.Container().
        From("cgr.dev/chainguard/wolfi-base:latest").
        WithDirectory("/opt/terranox/gnat", gnat).
        WithDirectory("/src", source).
        WithEnvVariable("PATH", "/opt/terranox/gnat/bin:/usr/bin:/bin").
        WithWorkdir("/src/kernel/ada").
        // Build ZFP runtime
        WithExec([]string{"gprbuild", "-P", "rts-zfp/zfp.gpr",
            "-XBUILD=production", "-j4"}).
        Directory("/src/kernel/ada/rts-zfp")
}

// BuildKernel compiles the full TerranoxOS kernel:
// C/ASM via Clang + Ada/SPARK via GNAT LLVM → lld → ELF
func (m *TerranoxBootstrap) BuildKernel(
    ctx context.Context,
    source *dagger.Directory,
    // +optional
    llvmVersion string,
) *dagger.File {
    cfg := m.cfg()
    ver := cfg.LLVMVersion(llvmVersion)
    stage1 := m.Stage1(ver)
    gnat := m.GnatLLVM(ctx, ver)

    return dag.Container().
        From("cgr.dev/chainguard/wolfi-base:latest").
        WithExec([]string{"apk", "add", "--no-cache",
            "bash", "make", "nasm"}).
        // Mount toolchains
        WithDirectory("/opt/terranox/tools", stage1).
        WithDirectory("/opt/terranox/gnat", gnat).
        WithDirectory("/src", source).
        WithEnvVariable("PATH",
            "/opt/terranox/tools/bin:/opt/terranox/gnat/bin:/usr/bin:/bin").
        WithWorkdir("/src").
        // Compile C sources
        WithExec([]string{"sh", "-c", `
            for f in kernel/core/*.c kernel/arch/x86_64/*.c kernel/boot/entry.c; do
                clang --target=x86_64-pc-terranox \
                    -ffreestanding -nostdlib -mno-red-zone \
                    -mno-sse -mcmodel=kernel -fno-pic \
                    -Ikernel/ -c "$f" -o "build/$(basename ${f%.c}.o)"
            done
        `}).
        // Compile ASM sources
        WithExec([]string{"sh", "-c", `
            for f in kernel/arch/x86_64/asm/*.S; do
                clang --target=x86_64-pc-terranox \
                    -c "$f" -o "build/$(basename ${f%.S}.o)"
            done
        `}).
        // Compile Ada sources
        WithExec([]string{"sh", "-c", `
            llvm-gnat --target=x86_64-pc-terranox -gnatwa \
                -c kernel/ada/subsystems/*.adb kernel/ada/drivers/*.adb \
                -o build/
        `}).
        // Link everything
        WithExec([]string{"ld.lld",
            "-T", "kernel/link/kernel.ld",
            "-nostdlib", "-static",
            "-o", "build/terranox-kernel.elf",
            "build/*.o"}).
        // Verify
        WithExec([]string{"file", "build/terranox-kernel.elf"}).
        WithExec([]string{"llvm-size", "build/terranox-kernel.elf"}).
        File("build/terranox-kernel.elf")
}

// MakeISO creates a bootable ISO with Limine + kernel.
func (m *TerranoxBootstrap) MakeISO(
    ctx context.Context,
    source *dagger.Directory,
    // +optional
    llvmVersion string,
) *dagger.File {
    kernel := m.BuildKernel(ctx, source, llvmVersion)

    return dag.Container().
        From("cgr.dev/chainguard/wolfi-base:latest").
        WithExec([]string{"apk", "add", "--no-cache",
            "xorriso", "mtools", "wget"}).
        // Get Limine
        WithExec([]string{"wget", "-q",
            "https://github.com/limine-bootloader/limine/releases/latest/download/limine-8.x-bin.tar.xz",
            "-O", "/tmp/limine.tar.xz"}).
        WithExec([]string{"tar", "xf", "/tmp/limine.tar.xz", "-C", "/tmp/"}).
        // Build ISO structure
        WithExec([]string{"mkdir", "-p", "/iso/boot"}).
        WithFile("/iso/boot/terranox-kernel.elf", kernel).
        WithDirectory("/src", source).
        WithExec([]string{"cp", "/src/kernel/boot/limine.conf", "/iso/boot/"}).
        // Create ISO
        WithExec([]string{"sh", "-c", `
            cp /tmp/limine-*/limine-bios.sys /iso/boot/
            cp /tmp/limine-*/limine-bios-cd.bin /iso/boot/
            cp /tmp/limine-*/limine-uefi-cd.bin /iso/boot/
            xorriso -as mkisofs \
                -b boot/limine-bios-cd.bin \
                -no-emul-boot -boot-load-size 4 -boot-info-table \
                --efi-boot boot/limine-uefi-cd.bin \
                -efi-boot-part --efi-boot-image \
                /iso -o /out/terranox.iso
            /tmp/limine-*/limine bios-install /out/terranox.iso
        `}).
        File("/out/terranox.iso")
}

// BootTest boots the ISO in QEMU headless and checks serial output.
func (m *TerranoxBootstrap) BootTest(
    ctx context.Context,
    source *dagger.Directory,
) (string, error) {
    iso := m.MakeISO(ctx, source, "")

    return dag.Container().
        From("cgr.dev/chainguard/wolfi-base:latest").
        WithExec([]string{"apk", "add", "--no-cache", "qemu-system-x86_64"}).
        WithFile("/tmp/terranox.iso", iso).
        // Boot with timeout, capture serial output
        WithExec([]string{"timeout", "30",
            "qemu-system-x86_64",
            "-cdrom", "/tmp/terranox.iso",
            "-serial", "file:/tmp/serial.txt",
            "-display", "none",
            "-m", "256M",
            "-no-reboot"}).
        // Check for expected output
        WithExec([]string{"cat", "/tmp/serial.txt"}).
        WithExec([]string{"grep", "-q", "terranox>", "/tmp/serial.txt"}).
        Stdout(ctx)
}
```
